cmake_minimum_required(VERSION 3.20)
project(usb_focus_control LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "USB focus control uses Windows DirectShow.")
endif()
add_executable(focus_control focus_control.cpp)
target_compile_features(focus_control PRIVATE cxx_std_17)
target_compile_definitions(focus_control PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(focus_control PRIVATE strmiids ole32 oleaut32)
